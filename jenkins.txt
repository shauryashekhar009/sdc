## ===== WEEK 9 : SCRIPTED PIPELINE (browser) =====
## New Item -> Scripted_Pipeline -> Pipeline -> OK
##   Description: Builds, tests and packages the Maven Java project from GitHub
##   Triggers: [x] Poll SCM -> H/5 * * * *
##   Pipeline -> Definition: Pipeline script -> paste below -> Save -> Build Now
pipeline {
  agent any
  tools { maven 'MAVEN-HOME' }
  stages {
    stage('git repo & clean') {
      steps {
        sh 'rm -rf simple-java-maven-app'
        sh 'git clone https://github.com/<your-username>/simple-java-maven-app.git'
        sh 'mvn clean -f simple-java-maven-app'
      }
    }
    stage('install') { steps { sh 'mvn install -f simple-java-maven-app' } }
    stage('test')    { steps { sh 'mvn test -f simple-java-maven-app' } }
    stage('package') { steps { sh 'mvn package -f simple-java-maven-app' } }
  }
}
